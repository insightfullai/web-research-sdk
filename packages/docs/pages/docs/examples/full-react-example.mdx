# Full React Example

This example demonstrates a complete React integration: provider setup, a custom modal study display, and event tracking through the `useInsightfull()` hook.

## Complete App

```tsx
import {
  InsightfullProvider,
  useInsightfull,
} from "@insightfull/web-research-sdk-react";
import {
  useCallback,
  useEffect,
  useRef,
  useState,
  type FormEvent,
} from "react";

// ── Custom Study Modal ──

function StudyModal({
  studyTitle,
  iframeUrl,
  onClose,
}: {
  studyTitle: string;
  iframeUrl: string;
  onClose: () => void;
}) {
  const backdropRef = useRef<HTMLDivElement>(null);

  // Close on Escape
  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      if (e.key === "Escape") onClose();
    };
    document.addEventListener("keydown", handler);
    return () => document.removeEventListener("keydown", handler);
  }, [onClose]);

  // Close on backdrop click
  const handleBackdropClick = useCallback(
    (e: React.MouseEvent) => {
      if (e.target === backdropRef.current) onClose();
    },
    [onClose],
  );

  // Lock body scroll
  useEffect(() => {
    document.body.style.overflow = "hidden";
    return () => {
      document.body.style.overflow = "";
    };
  }, []);

  return (
    <div
      ref={backdropRef}
      onClick={handleBackdropClick}
      style={{
        position: "fixed",
        inset: 0,
        zIndex: 999998,
        display: "flex",
        alignItems: "center",
        justifyContent: "center",
        background: "rgba(0, 0, 0, 0.6)",
        backdropFilter: "blur(4px)",
      }}
    >
      <div
        style={{
          position: "relative",
          width: "90vw",
          maxWidth: 600,
          height: "80vh",
          maxHeight: 750,
          background: "#fff",
          borderRadius: 16,
          boxShadow: "0 20px 60px rgba(0, 0, 0, 0.2)",
          overflow: "hidden",
        }}
      >
        <button
          onClick={onClose}
          aria-label="Close study"
          style={{
            position: "absolute",
            top: 12,
            right: 12,
            zIndex: 100,
            display: "flex",
            alignItems: "center",
            justifyContent: "center",
            width: 32,
            height: 32,
            border: "none",
            borderRadius: "50%",
            background: "rgba(0, 0, 0, 0.6)",
            color: "#fff",
            fontSize: 20,
            lineHeight: 1,
            cursor: "pointer",
          }}
        >
          &#xd7;
        </button>
        <iframe
          src={iframeUrl}
          title={studyTitle}
          allow="clipboard-write"
          style={{
            width: "100%",
            height: "100%",
            border: "none",
          }}
        />
      </div>
    </div>
  );
}

// ── App Content ──

function App() {
  const { sdk, isReady } = useInsightfull();

  const handleTrack = useCallback(
    (eventName: string) => {
      if (isReady && sdk) {
        sdk.track(eventName);
      }
    },
    [sdk, isReady],
  );

  return (
    <div style={{ padding: 40, fontFamily: "system-ui, sans-serif" }}>
      <h1>My Application</h1>
      <div style={{ display: "flex", gap: 12 }}>
        <button onClick={() => handleTrack("pricing_viewed")}>
          View Pricing
        </button>
        <button onClick={() => handleTrack("demo_requested")}>
          Request Demo
        </button>
        <button onClick={() => handleTrack("checkout_started")}>
          Start Checkout
        </button>
      </div>
    </div>
  );
}

// ── Root ──

export default function Root() {
  const [activeStudy, setActiveStudy] = useState<{
    title: string;
    iframeUrl: string;
  } | null>(null);

  const handleStudyTrigger = useCallback(
    (params: {
      study: { title: string | null; id: number };
      iframeUrl: string;
    }) => {
      setActiveStudy({
        title: params.study.title ?? `Study ${params.study.id}`,
        iframeUrl: params.iframeUrl,
      });
    },
    [],
  );

  return (
    <InsightfullProvider clientId="env_abc" onStudyTrigger={handleStudyTrigger}>
      <App />
      {activeStudy && (
        <StudyModal
          studyTitle={activeStudy.title}
          iframeUrl={activeStudy.iframeUrl}
          onClose={() => setActiveStudy(null)}
        />
      )}
    </InsightfullProvider>
  );
}
```

## Key patterns

### Provider at the root

```tsx
<InsightfullProvider
  clientId="env_abc"
  onStudyTrigger={({ iframeUrl }) => setActiveStudy(iframeUrl)}
>
  <App />
</InsightfullProvider>
```

The `onStudyTrigger` prop receives `{ study, iframeUrl, context }`. Use the `iframeUrl` directly — it already includes the base64-encoded context payload. Store it in state and conditionally render your modal.

### Guard SDK calls with `isReady`

```tsx
const { sdk, isReady } = useInsightfull();

if (isReady && sdk) {
  sdk.track("pricing_viewed");
}
```

During SSR and the initial client render, `sdk` is `null` and `isReady` is `false`. Always guard before calling SDK methods.

### Modal as a controlled component

The modal is rendered conditionally based on React state. When the user closes it, state is set to `null` and the component unmounts. The `useEffect` in `StudyModal` restores body scroll on unmount.

### Escape and backdrop close

- **Escape key** — a `keydown` listener on `document` with a cleanup return.
- **Backdrop click** — an `onClick` handler that checks `e.target === backdropRef.current` so clicks inside the modal are not intercepted.

### Cleanup

The `<InsightfullProvider>` does not auto-call `sdk.destroy()` on unmount. If you need cleanup when the component tree unmounts, add a wrapping `useEffect`:

```tsx
function Main() {
  const { sdk } = useInsightfull();

  useEffect(() => {
    return () => {
      sdk?.destroy();
    };
  }, [sdk]);

  return <App />;
}
```
